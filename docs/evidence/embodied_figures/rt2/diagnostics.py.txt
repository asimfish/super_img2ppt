def mask(im,b,col):
 crop=im.crop(b).convert('RGB'); vals=[]
 for rr,g,bb in crop.getdata():
  if col=='gray':v=max(rr,g,bb)-min(rr,g,bb)<22 and max(rr,g,bb)<170
  elif col=='purple':v=bb>rr+20 and rr>g+8 and rr<170 and g<150
  else:v=rr>230 and 70<g<190 and bb<80
  vals.append(255 if v else 0)
 out=Image.new('L',crop.size);out.putdata(vals);return out

def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()

 out=Image.open(r/'build_02/actual_source_width.png');rows=[]

  sb,ob=s.getbbox(),o.getbbox();sp=list(s.getdata());op=list(o.getdata());intersection=sum(bool(a and b) for a,b in zip(sp,op));union=sum(bool(a or b) for a,b in zip(sp,op))
  rows.append(dict(split=split,id=name,source_bounds=sb,actual_bounds=ob,edge_delta_px=[v-u for u,v in zip(sb,ob)],iou=intersection/union,source_mask_validated=True))
