def sh(id,box,fill=None,stroke=None,sw=2,round=0,container=None,z=1):
 e=dict(id=id,kind='shape',z=z,box=box,shape='round_rect' if round else 'rect',fill=fill,stroke=stroke,stroke_width=sw)

 es.append(e);return e

def text(id,txt,box,size=22,color=gray,bold=False,align='left',container=None,z=10,runs=None):
 e=dict(id=id,kind='text',z=z,box=box,font_size=size,font_family='Arial',color=color,bold=bold,align=align,valign='top',fit='strict',wrap=False)
 e['runs' if runs else 'text']=runs if runs else txt
