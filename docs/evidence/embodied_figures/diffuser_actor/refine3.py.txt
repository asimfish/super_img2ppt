  for j,r in enumerate([3.6,5.8]):
   for sign in [-1,1]:
    a2=a+sign*math.pi/3
    ln(f'{pre}_{k}_{j}_{sign}',[cx+r*ux,cy+r*uy],[cx+r*ux+2.3*math.cos(a2),cy+r*uy+2.3*math.sin(a2)],1.1)
