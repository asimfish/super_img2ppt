def ink(im,b):
 px=im.crop(b);coords=[]
 for y in range(px.height):
  for x in range(px.width):
   rgb=px.getpixel((x,y))
   if max(rgb)<170 or (max(rgb)-min(rgb)>45 and min(rgb)<170): coords.append((x+b[0],y+b[1]))
 if not coords:return None
 return {'bbox':[min(a[0] for a in coords),min(a[1] for a in coords),max(a[0] for a in coords)+1,max(a[1] for a in coords)+1],'centroid':[sum(a[k] for a in coords)/len(coords) for k in [0,1]],'count':len(coords)}
