def mask(im,a):
 l,t,r,b=a['region'];res=set();kind=a['kind'];bg=a.get('background_rgb',[255,255,255])
 for y in range(t,b):
  for x in range(l,r):
   rgb=im.getpixel((x,y));okay=False
   if kind in ['text','text_raster']: okay=max(abs(rgb[k]-bg[k]) for k in range(3))>=70
   elif 'grid' in kind:okay=min(rgb)>=250
   elif kind=='outline':okay=max(rgb)<=80 and max(rgb)-min(rgb)<=8
   if okay:res.add((x,y))
 return res

def bbox(m):return [min(x for x,y in m),min(y for x,y in m),max(x for x,y in m)+1,max(y for x,y in m)+1] if m else None
